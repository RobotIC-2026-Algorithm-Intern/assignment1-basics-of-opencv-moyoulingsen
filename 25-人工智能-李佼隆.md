# 根据HSV值分类颜色
def classify_color(h, s, v):
    # 根据色调(H)判断颜色
    if s < 50 or v < 50:  # 饱和度或亮度太低，可能是灰色或黑色
        if v < 30:
            return "Black"
        elif s < 30:
            return "White" if v > 200 else "Gray"
    
    if (0 <= h <= 10) or (170 <= h <= 180):
        return "Red"
    elif 11 <= h <= 25:
        return "Orange"
    elif 26 <= h <= 35:
        return "Yellow"
    elif 36 <= h <= 85:
        return "Green"
    elif 86 <= h <= 125:
        return "Blue"
    elif 126 <= h <= 155:
        return "Purple"
    elif 156 <= h <= 169:
        return "Pink"
    
    return "Unknown"

# 计算ROI中各颜色的分布
def calculate_color_distribution(hsv_roi):
    height, width = hsv_roi.shape[:2]
    total_pixels = height * width
    
    # 如果区域太大，抽样处理以提高性能
    if total_pixels > 10000:
        step = int(np.sqrt(total_pixels / 1000))
        hsv_roi = hsv_roi[::step, ::step]
        height, width = hsv_roi.shape[:2]
        total_pixels = height * width
    
    color_count = {}
    
    # 对每个像素进行分类
    for y in range(height):
        for x in range(width):
            h, s, v = hsv_roi[y, x]
            color = classify_color(h, s, v)
            color_count[color] = color_count.get(color, 0) + 1
    
    # 计算百分比
    color_distribution = {}
    for color, count in color_count.items():
        color_distribution[color] = (count / total_pixels) * 100
    
    return color_distribution

import cv2
import numpy as np
from collections import Counter
from typing import List, Tuple, Dict, Optional, Union

class HSVColorProcessor:
    """
    增强版 HSV 颜色处理器（带颜色名称识别）
    功能：
    1. 识别图像中 HSV 颜色最多的部分
    2. 分析所有颜色的成分比例，并显示对应颜色名称
    3. 支持添加和管理多个颜色范围
    4. 生成单个或多个颜色范围的掩膜
    5. 删除或保留指定颜色范围
    6. 预定义常见颜色范围
    """
    
    def __init__(self, h_bins: int = 30, s_bins: int = 32, v_bins: int = 32):
        """
        初始化 HSV 颜色处理器
        
        参数:
            h_bins: H 通道的直方图 bin 数量
            s_bins: S 通道的直方图 bin 数量
            v_bins: V 通道的直方图 bin 数量
        """
        self.h_bins = h_bins
        self.s_bins = s_bins
        self.v_bins = v_bins
        self.h_ranges = [0, 180]  # OpenCV 中 H 通道范围是 0-179
        self.s_ranges = [0, 256]  # S 通道范围是 0-255
        self.v_ranges = [0, 256]  # V 通道范围是 0-255
        
        self.color_ranges = {}  # 存储颜色范围，格式: {name: [(lower, upper), ...]}
        
        # 预定义常见颜色
        self._setup_common_colors()
        
        # 颜色名称映射
        self.color_names = {
            "red": "红色",
            "green": "绿色", 
            "blue": "蓝色",
            "yellow": "黄色",
            "orange": "橙色",
            "purple": "紫色",
            "pink": "粉色",
            "white": "白色",
            "black": "黑色",
            "gray": "灰色"
        }
    
    def _setup_common_colors(self):
        """
        设置常见颜色的 HSV 范围
        """
        # 红色（两个范围）
        red_ranges = [
            ((0, 100, 100), (10, 255, 255)),
            ((170, 100, 100), (180, 255, 255))
        ]
        self.add_color_ranges("red", red_ranges)
        
        # 绿色
        self.add_color_range("green", (40, 100, 100), (80, 255, 255))
        
        # 蓝色
        self.add_color_range("blue", (100, 100, 100), (130, 255, 255))
        
        # 黄色
        self.add_color_range("yellow", (20, 100, 100), (30, 255, 255))
        
        # 橙色
        self.add_color_range("orange", (10, 100, 100), (20, 255, 255))
        
        # 紫色
        self.add_color_range("purple", (130, 100, 100), (160, 255, 255))
        
        # 粉色
        self.add_color_range("pink", (140, 50, 100), (170, 255, 255))
        
        # 白色
        self.add_color_range("white", (0, 0, 200), (180, 50, 255))
        
        # 黑色
        self.add_color_range("black", (0, 0, 0), (180, 255, 50))
        
        # 灰色
        self.add_color_range("gray", (0, 0, 50), (180, 50, 200))
    
    def hsv_to_color_name(self, h: int, s: int, v: int) -> str:
        """
        将 HSV 值转换为颜色名称
        
        参数:
            h: 色调 (0-179)
            s: 饱和度 (0-255)
            v: 亮度 (0-255)
            
        返回:
            颜色名称
        """
        # 根据 HSV 值判断颜色
        if v < 50:
            return "黑色"
        elif s < 50 and v > 200:
            return "白色"
        elif s < 50 and v > 50:
            return "灰色"
        elif h < 10 or h > 170:
            return "红色"
        elif h < 20:
            return "橙色"
        elif h < 30:
            return "黄色"
        elif h < 40:
            return "黄绿色"
        elif h < 80:
            return "绿色"
        elif h < 100:
            return "青绿色"
        elif h < 130:
            return "蓝色"
        elif h < 160:
            return "紫色"
        else:
            return "粉色"
    
    def add_color_range(self, name: str, 
                       lower_hsv: Tuple[int, int, int], 
                       upper_hsv: Tuple[int, int, int]) -> None:
        """
        添加单个颜色范围
        
        参数:
            name: 颜色范围的名称（如 "red", "blue"）
            lower_hsv: HSV 下限 (H, S, V)
            upper_hsv: HSV 上限 (H, S, V)
        """
        if name not in self.color_ranges:
            self.color_ranges[name] = []
        
        self.color_ranges[name].append((lower_hsv, upper_hsv))
    
    def add_color_ranges(self, name: str, 
                        ranges: List[Tuple[Tuple[int, int, int], Tuple[int, int, int]]]) -> None:
        """
        添加多个颜色范围（例如红色的两个范围）
        
        参数:
            name: 颜色范围的名称
            ranges: 颜色范围列表，每个元素为 (lower_hsv, upper_hsv)
        """
        if name not in self.color_ranges:
            self.color_ranges[name] = []
        
        self.color_ranges[name].extend(ranges)
    
    def remove_color_range(self, name: str) -> bool:
        """
        移除指定的颜色范围
        
        参数:
            name: 要移除的颜色范围名称
            
        返回:
            成功移除返回 True，未找到返回 False
        """
        if name in self.color_ranges:
            del self.color_ranges[name]
            return True
        return False
    
    def get_color_mask(self, hsv_image: np.ndarray, 
                      color_name: str, 
                      operation: str = "union") -> np.ndarray:
        """
        生成指定颜色名称的掩膜
        
        参数:
            hsv_image: HSV 格式的图像
            color_name: 颜色名称
            operation: 多个颜色范围的操作方式，"union"（并集）或 "intersection"（交集）
            
        返回:
            颜色掩膜（二值图像）
        """
        if color_name not in self.color_ranges:
            raise ValueError(f"未找到颜色范围: {color_name}")
        
        ranges = self.color_ranges[color_name]
        
        if operation == "union":
            # 多个范围的并集
            mask = np.zeros(hsv_image.shape[:2], dtype=np.uint8)
            for lower, upper in ranges:
                range_mask = cv2.inRange(hsv_image, np.array(lower), np.array(upper))
                mask = cv2.bitwise_or(mask, range_mask)
        elif operation == "intersection":
            # 多个范围的交集
            mask = np.ones(hsv_image.shape[:2], dtype=np.uint8) * 255
            for lower, upper in ranges:
                range_mask = cv2.inRange(hsv_image, np.array(lower), np.array(upper))
                mask = cv2.bitwise_and(mask, range_mask)
        else:
            raise ValueError("operation 必须是 'union' 或 'intersection'")
        
        return mask
    
    def get_combined_mask(self, hsv_image: np.ndarray, 
                         color_names: List[str],
                         operation: str = "union") -> np.ndarray:
        """
        生成多个颜色组合的掩膜
        
        参数:
            hsv_image: HSV 格式的图像
            color_names: 颜色名称列表
            operation: 多个颜色之间的操作方式，"union"（并集）或 "intersection"（交集）
            
        返回:
            组合颜色掩膜（二值图像）
        """
        if not color_names:
            raise ValueError("颜色名称列表不能为空")
        
        masks = []
        for name in color_names:
            if name not in self.color_ranges:
                raise ValueError(f"未找到颜色范围: {name}")
            mask = self.get_color_mask(hsv_image, name, "union")
            masks.append(mask)
        
        if operation == "union":
            # 多个颜色的并集
            combined_mask = np.zeros(hsv_image.shape[:2], dtype=np.uint8)
            for mask in masks:
                combined_mask = cv2.bitwise_or(combined_mask, mask)
        elif operation == "intersection":
            # 多个颜色的交集
            combined_mask = np.ones(hsv_image.shape[:2], dtype=np.uint8) * 255
            for mask in masks:
                combined_mask = cv2.bitwise_and(combined_mask, mask)
        else:
            raise ValueError("operation 必须是 'union' 或 'intersection'")
        
        return combined_mask
    
    def get_color_distribution(self, hsv_image: np.ndarray) -> Tuple[np.ndarray, Dict]:
        """
        获取 HSV 图像的颜色分布
        
        参数:
            hsv_image: HSV 格式的图像
            
        返回:
            hist: 3D 直方图
            color_stats: 颜色分布统计信息
        """
        # 计算 3D 直方图
        hist = cv2.calcHist([hsv_image], [0, 1, 2], None, 
                           [self.h_bins, self.s_bins, self.v_bins], 
                           self.h_ranges + self.s_ranges + self.v_ranges)
        
        # 计算每个像素的量化颜色索引
        h_quantized = (hsv_image[:, :, 0] / (180 / self.h_bins)).astype(int)
        s_quantized = (hsv_image[:, :, 1] / (256 / self.s_bins)).astype(int)
        v_quantized = (hsv_image[:, :, 2] / (256 / self.v_bins)).astype(int)
        
        # 限制在有效范围内
        h_quantized = np.clip(h_quantized, 0, self.h_bins - 1)
        s_quantized = np.clip(s_quantized, 0, self.s_bins - 1)
        v_quantized = np.clip(v_quantized, 0, self.v_bins - 1)
        
        # 统计颜色分布
        color_indices = list(zip(h_quantized.flatten(), s_quantized.flatten(), v_quantized.flatten()))
        color_counter = Counter(color_indices)
        
        # 计算总像素数
        total_pixels = hsv_image.shape[0] * hsv_image.shape[1]
        
        # 转换为比例
        color_stats = {}
        for color, count in color_counter.items():
            color_stats[color] = count / total_pixels
        
        return hist, color_stats
    
    def get_dominant_colors(self, hsv_image: np.ndarray, top_k: int = 5) -> List[Tuple[Tuple[int, int, int], float, str]]:
        """
        获取图像中占比最多的几种颜色及其比例和颜色名称
        
        参数:
            hsv_image: HSV 格式的图像
            top_k: 返回前几种主要颜色
            
        返回:
            颜色列表，每个元素为 (HSV 值, 比例, 颜色名称)
        """
        _, color_stats = self.get_color_distribution(hsv_image)
        
        # 按比例排序
        sorted_colors = sorted(color_stats.items(), key=lambda x: x[1], reverse=True)
        
        # 转换为实际的 HSV 值和颜色名称
        dominant_colors = []
        for (h_idx, s_idx, v_idx), proportion in sorted_colors[:top_k]:
            h_val = int((h_idx + 0.5) * (180 / self.h_bins))
            s_val = int((s_idx + 0.5) * (256 / self.s_bins))
            v_val = int((v_idx + 0.5) * (256 / self.v_bins))
            color_name = self.hsv_to_color_name(h_val, s_val, v_val)
            dominant_colors.append(((h_val, s_val, v_val), proportion, color_name))
        
        return dominant_colors
    
    def get_all_color_proportions(self, hsv_image: np.ndarray, threshold: float = 0.01) -> List[Tuple[Tuple[int, int, int], float, str]]:
        """
        获取图像中所有颜色成分的比例（超过阈值）
        
        参数:
            hsv_image: HSV 格式的图像
            threshold: 显示比例阈值，低于此值的不显示
            
        返回:
            颜色列表，每个元素为 (HSV 值, 比例, 颜色名称)
        """
        _, color_stats = self.get_color_distribution(hsv_image)
        
        # 筛选超过阈值的颜色
        filtered_colors = [(color, prop) for color, prop in color_stats.items() if prop >= threshold]
        
        # 按比例排序
        sorted_colors = sorted(filtered_colors, key=lambda x: x[1], reverse=True)
        
        # 转换为实际的 HSV 值和颜色名称
        color_proportions = []
        for (h_idx, s_idx, v_idx), proportion in sorted_colors:
            h_val = int((h_idx + 0.5) * (180 / self.h_bins))
            s_val = int((s_idx + 0.5) * (256 / self.s_bins))
            v_val = int((v_idx + 0.5) * (256 / self.v_bins))
            color_name = self.hsv_to_color_name(h_val, s_val, v_val)
            color_proportions.append(((h_val, s_val, v_val), proportion, color_name))
        
        return color_proportions
    
    def remove_colors(self, hsv_image: np.ndarray, 
                     color_names: Union[str, List[str]]) -> np.ndarray:
        """
        删除指定颜色（将其变为黑色）
        
        参数:
            hsv_image: HSV 格式的图像
            color_names: 颜色名称或颜色名称列表
            
        返回:
            处理后的 HSV 图像
        """
        if isinstance(color_names, str):
            color_names = [color_names]
        
        mask = self.get_combined_mask(hsv_image, color_names, "union")
        
        # 将目标颜色区域变为黑色
        result = hsv_image.copy()
        result[mask > 0] = [0, 0, 0]  # HSV 中的黑色是 (0, 0, 0)
        
        return result
    
    def keep_colors(self, hsv_image: np.ndarray, 
                   color_names: Union[str, List[str]]) -> np.ndarray:
        """
        只保留指定颜色（其他变为黑色）
        
        参数:
            hsv_image: HSV 格式的图像
            color_names: 颜色名称或颜色名称列表
            
        返回:
            处理后的 HSV 图像
        """
        if isinstance(color_names, str):
            color_names = [color_names]
        
        mask = self.get_combined_mask(hsv_image, color_names, "union")
        
        # 将非目标颜色区域变为黑色
        result = hsv_image.copy()
        result[mask == 0] = [0, 0, 0]  # HSV 中的黑色是 (0, 0, 0)
        
        return result
    
    def remove_color_by_value(self, hsv_image: np.ndarray, target_hsv: Tuple[int, int, int], 
                             tolerance: Tuple[int, int, int] = (10, 50, 50)) -> np.ndarray:
        """
        删除指定 HSV 值范围的颜色（将其变为黑色）
        
        参数:
            hsv_image: HSV 格式的图像
            target_hsv: 目标 HSV 颜色 (H, S, V)
            tolerance: 容忍度 (H_tol, S_tol, V_tol)
            
        返回:
            处理后的 HSV 图像
        """
        h, s, v = target_hsv
        h_tol, s_tol, v_tol = tolerance
        
        # 创建掩码
        lower_bound = np.array([max(0, h - h_tol), max(0, s - s_tol), max(0, v - v_tol)])
        upper_bound = np.array([min(179, h + h_tol), min(255, s + s_tol), min(255, v + v_tol)])
        
        mask = cv2.inRange(hsv_image, lower_bound, upper_bound)
        
        # 将目标颜色区域变为黑色
        result = hsv_image.copy()
        result[mask > 0] = [0, 0, 0]  # HSV 中的黑色是 (0, 0, 0)
        
        return result
    
    def keep_color_by_value(self, hsv_image: np.ndarray, target_hsv: Tuple[int, int, int],

        """

    def apply_mask(self, hsv_image: np.ndarray, 
                  mask: np.ndarray, 
                  inverse: bool = False) -> np.ndarray:
        """
        应用掩膜到图像
        
        参数:
            hsv_image: HSV 格式的图像
            mask: 二值掩膜
            inverse: 是否反转掩膜
            
        返回:

        if inverse:

        result = hsv_image.copy()
        result[mask == 0] = [0, 0, 0]
        
        return result

    def add_color_ranges(self, name: str,

    def remove_color_range(self, name: str) -> bool:
        """

    def get_color_mask(self, hsv_image: np.ndarray, 
                      color_name: str, 
                      operation: str = "union") -> np.ndarray:
        """
        生成指定颜色名称的掩膜
        
        参数:
            hsv_image: HSV 格式的图像
            color_name: 颜色名称
            operation: 多个颜色范围的操作方式，"union"（并集）或 "intersection"（交集）
            
        返回:
            颜色掩膜（二值图像）
        """

            raise ValueError(f"未找到颜色范围: {color_name}")
        
        ranges = self.color_ranges[color_name]
        
        if operation == "union":
            # 多个范围的并集
            mask = np.zeros(hsv_image.shape[:2], dtype=np.uint8)
            for lower, upper in ranges:
                range_mask = cv2.inRange(hsv_image, np.array(lower), np.array(upper))
                mask = cv2.bitwise_or(mask, range_mask)
        elif operation == "intersection":
            # 多个范围的交集
            mask = np.ones(hsv_image.shape[:2], dtype=np.uint8) * 255
            for lower, upper in ranges:
                range_mask = cv2.inRange(hsv_image, np.array(lower), np.array(upper))
                mask = cv2.bitwise_and(mask, range_mask)
        else:
            raise ValueError("operation 必须是 'union' 或 'intersection'")
        
        return mask
    
    def get_combined_mask(self, hsv_image: np.ndarray, 
                         color_names: List[str],
                         operation: str = "union") -> np.ndarray:
        """
        生成多个颜色组合的掩膜
        
        参数:
            hsv_image: HSV 格式的图像
            color_names: 颜色名称列表
            operation: 多个颜色之间的操作方式，"union"（并集）或 "intersection"（交集）
            
        返回:
            组合颜色掩膜（二值图像）
        """
        if not color_names:
            raise ValueError("颜色名称列表不能为空")
        
        masks = []
        for name in color_names:
            if name not in self.color_ranges:
                raise ValueError(f"未找到颜色范围: {name}")
            mask = self.get_color_mask(hsv_image, name, "union")
            masks.append(mask)
        
        if operation == "union":
            # 多个颜色的并集
            combined_mask = np.zeros(hsv_image.shape[:2], dtype=np.uint8)
            for mask in masks:
                combined_mask = cv2.bitwise_or(combined_mask, mask)
        elif operation == "intersection":
            # 多个颜色的交集
            combined_mask = np.ones(hsv_image.shape[:2], dtype=np.uint8) * 255
            for mask in masks:
                combined_mask = cv2.bitwise_and(combined_mask, mask)
        else:
            raise ValueError("operation 必须是 'union' 或 'intersection'")
        
        return combined_mask
    
    def get_color_distribution(self, hsv_image: np.ndarray) -> Tuple[np.ndarray, Dict]:
        """
        获取 HSV 图像的颜色分布
        
        参数:
            hsv_image: HSV 格式的图像
            
        返回:
            hist: 3D 直方图
            color_stats: 颜色分布统计信息
        """
        # 计算 3D 直方图
        hist = cv2.calcHist([hsv_image], [0, 1, 2], None, 
                           [self.h_bins, self.s_bins, self.v_bins], 
                           self.h_ranges + self.s_ranges + self.v_ranges)
        
        # 计算每个像素的量化颜色索引
        h_quantized = (hsv_image[:, :, 0] / (180 / self.h_bins)).astype(int)
        s_quantized = (hsv_image[:, :, 1] / (256 / self.s_bins)).astype(int)
        v_quantized = (hsv_image[:, :, 2] / (256 / self.v_bins)).astype(int)
        
        # 限制在有效范围内
        h_quantized = np.clip(h_quantized, 0, self.h_bins - 1)
        s_quantized = np.clip(s_quantized, 0, self.s_bins - 1)
        v_quantized = np.clip(v_quantized, 0, self.v_bins - 1)
        
        # 统计颜色分布
        color_indices = list(zip(h_quantized.flatten(), s_quantized.flatten(), v_quantized.flatten()))
        color_counter = Counter(color_indices)
        
        # 计算总像素数
        total_pixels = hsv_image.shape[0] * hsv_image.shape[1]
        
        # 转换为比例
        color_stats = {}
        for color, count in color_counter.items():
            color_stats[color] = count / total_pixels
        
        return hist, color_stats
    
    def get_dominant_colors(self, hsv_image: np.ndarray, top_k: int = 5) -> List[Tuple[Tuple[int, int, int], float, str]]:
        """
        获取图像中占比最多的几种颜色及其比例和颜色名称
        
        参数:
            hsv_image: HSV 格式的图像
            top_k: 返回前几种主要颜色
            
        返回:
            颜色列表，每个元素为 (HSV 值, 比例, 颜色名称)
        """
        _, color_stats = self.get_color_distribution(hsv_image)
        
        # 按比例排序
        sorted_colors = sorted(color_stats.items(), key=lambda x: x[1], reverse=True)
        
        # 转换为实际的 HSV 值和颜色名称
        dominant_colors = []
        for (h_idx, s_idx, v_idx), proportion in sorted_colors[:top_k]:
            h_val = int((h_idx + 0.5) * (180 / self.h_bins))
            s_val = int((s_idx + 0.5) * (256 / self.s_bins))
            v_val = int((v_idx + 0.5) * (256 / self.v_bins))
            color_name = self.hsv_to_color_name(h_val, s_val, v_val)
            dominant_colors.append(((h_val, s_val, v_val), proportion, color_name))
        
        return dominant_colors
    
    def get_all_color_proportions(self, hsv_image: np.ndarray, threshold: float = 0.01) -> List[Tuple[Tuple[int, int, int], float, str]]:
        """
        获取图像中所有颜色成分的比例（超过阈值）
        
        参数:
            hsv_image: HSV 格式的图像
            threshold: 显示比例阈值，低于此值的不显示
            
        返回:
            颜色列表，每个元素为 (HSV 值, 比例, 颜色名称)
        """
        _, color_stats = self.get_color_distribution(hsv_image)
        
        # 筛选超过阈值的颜色
        filtered_colors = [(color, prop) for color, prop in color_stats.items() if prop >= threshold]
        
        # 按比例排序
        sorted_colors = sorted(filtered_colors, key=lambda x: x[1], reverse=True)
        
        # 转换为实际的 HSV 值和颜色名称
        color_proportions = []
        for (h_idx, s_idx, v_idx), proportion in sorted_colors:
            h_val = int((h_idx + 0.5) * (180 / self.h_bins))
            s_val = int((s_idx + 0.5) * (256 / self.s_bins))
            v_val = int((v_idx + 0.5) * (256 / self.v_bins))
            color_name = self.hsv_to_color_name(h_val, s_val, v_val)
            color_proportions.append(((h_val, s_val, v_val), proportion, color_name))
        
        return color_proportions
    
    def remove_colors(self, hsv_image: np.ndarray, 
                     color_names: Union[str, List[str]]) -> np.ndarray:
        """
        删除指定颜色（将其变为黑色）
        
        参数:
            hsv_image: HSV 格式的图像
            color_names: 颜色名称或颜色名称列表
            
        返回:
            处理后的 HSV 图像
        """
        if isinstance(color_names, str):
            color_names = [color_names]
        
        mask = self.get_combined_mask(hsv_image, color_names, "union")
        
        # 将目标颜色区域变为黑色
        result = hsv_image.copy()
        result[mask > 0] = [0, 0, 0]  # HSV 中的黑色是 (0, 0, 0)
        
        return result
    
    def keep_colors(self, hsv_image: np.ndarray, 
                   color_names: Union[str, List[str]]) -> np.ndarray:
        """
        只保留指定颜色（其他变为黑色）
        
        参数:
            hsv_image: HSV 格式的图像
            color_names: 颜色名称或颜色名称列表
            
        返回:
            处理后的 HSV 图像
        """
        if isinstance(color_names, str):
            color_names = [color_names]
        
        mask = self.get_combined_mask(hsv_image, color_names, "union")
        
        # 将非目标颜色区域变为黑色
        result = hsv_image.copy()
        result[mask == 0] = [0, 0, 0]  # HSV 中的黑色是 (0, 0, 0)
        
        return result
    
    def remove_color_by_value(self, hsv_image: np.ndarray, target_hsv: Tuple[int, int, int], 
                             tolerance: Tuple[int, int, int] = (10, 50, 50)) -> np.ndarray:
        """
        删除指定 HSV 值范围的颜色（将其变为黑色）
        
        参数:
            hsv_image: HSV 格式的图像
            target_hsv: 目标 HSV 颜色 (H, S, V)
            tolerance: 容忍度 (H_tol, S_tol, V_tol)
            
        返回:

        """
        h, s, v = target_hsv
        h_tol, s_tol, v_tol = tolerance
        
        # 创建掩码
        lower_bound = np.array([max(0, h - h_tol), max(0, s - s_tol), max(0, v - v_tol)])
        upper_bound = np.array([min(179, h + h_tol), min(255, s + s_tol), min(255, v + v_tol)])
        
        mask = cv2.inRange(hsv_image, lower_bound, upper_bound)
        
        # 将目标颜色区域变为黑色
        result = hsv_image.copy()
        result[mask > 0] = [0, 0, 0]  # HSV 中的黑色是 (0, 0, 0)
        
        return result
    
    def keep_color_by_value(self, hsv_image: np.ndarray, target_hsv: Tuple[int, int, int],

        """

    def apply_mask(self, hsv_image: np.ndarray, 
                  mask: np.ndarray, 
                  inverse: bool = False) -> np.ndarray:
        """
        应用掩膜到图像
        
        参数:
            hsv_image: HSV 格式的图像
            mask: 二值掩膜
            inverse: 是否反转掩膜
            
        返回:

        if inverse:
            mask = cv2.bitwise_not(mask)
        
        result = hsv_image.copy()
        result[mask == 0] = [0, 0, 0]
        
        return result

class RGBColorProcessor:

    def rgb_to_color_name(self, r: int, g: int, b: int) -> str:
        """
        将 RGB 值转换为颜色名称
        
        参数:
            r: 红色分量 (0-255)
            g: 绿色分量 (0-255)
            b: 蓝色分量 (0-255)
            
        返回:
            颜色名称
        """
        # 计算颜色的亮度和饱和度
        max_val = max(r, g, b)
        min_val = min(r, g, b)
        delta = max_val - min_val
        
        # 判断黑色、白色和灰色
        if max_val < 60:
            return "黑色"
        elif min_val > 200:
            return "白色"
        elif delta < 30:
            return "灰色"
        
        # 判断彩色
        if r > max(g, b) and r - min(g, b) > 50:
            if g > 150 and r > 200 and b < 100:
                return "橙色"
            elif g > 150 and b < 100:
                return "黄色"
            elif g < 100 and b < 100:
                return "红色"
            else:
                return "粉色"
        elif g > max(r, b) and g - min(r, b) > 50:
            return "绿色"
        elif b > max(r, g) and b - min(r, g) > 50:
            if r > 150 and b > 150 and g < 100:
                return "紫色"
            else:
                return "蓝色"
        elif g > 150 and b > 150 and r < 100:
            return "青色"
        elif r > 100 and g > 50 and b < 50:
            return "棕色"
        elif r > 150 and b > 150 and g < 100:
            return "品红色"
        else:
            return "混合色"
    
    def add_color_range(self, name: str, 
                       lower_rgb: Tuple[int, int, int], 
                       upper_rgb: Tuple[int, int, int]) -> None:
        """
        添加单个颜色范围
        
        参数:
            name: 颜色范围的名称（如 "red", "blue"）
            lower_rgb: RGB 下限 (R, G, B)
            upper_rgb: RGB 上限 (R, G, B)
        """
        if name not in self.color_ranges:
            self.color_ranges[name] = []
        
        self.color_ranges[name].append((lower_rgb, upper_rgb))
    
    def add_color_ranges(self, name: str, 
                        ranges: List[Tuple[Tuple[int, int, int], Tuple[int, int, int]]]) -> None:
        """
        添加多个颜色范围（例如红色的两个范围）
        
        参数:
            name: 颜色范围的名称
            ranges: 颜色范围列表，每个元素为 (lower_rgb, upper_rgb)
        """
        if name not in self.color_ranges:
            self.color_ranges[name] = []
        
        self.color_ranges[name].extend(ranges)
    
    def remove_color_range(self, name: str) -> bool:
        """
        移除指定的颜色范围
        
        参数:
            name: 要移除的颜色范围名称
            
        返回:
            成功移除返回 True，未找到返回 False
        """
        if name in self.color_ranges:
            del self.color_ranges[name]
            return True
        return False
    
    def get_color_mask(self, rgb_image: np.ndarray, 
                      color_name: str, 
                      operation: str = "union") -> np.ndarray:
        """
        生成指定颜色名称的掩膜
        
        参数:
            rgb_image: RGB 格式的图像
            color_name: 颜色名称
            operation: 多个颜色范围的操作方式，"union"（并集）或 "intersection"（交集）
            
        返回:
            颜色掩膜（二值图像）
        """
        if color_name not in self.color_ranges:
            raise ValueError(f"未找到颜色范围: {color_name}")
        
        ranges = self.color_ranges[color_name]
        
        if operation == "union":
            # 多个范围的并集
            mask = np.zeros(rgb_image.shape[:2], dtype=np.uint8)
            for lower, upper in ranges:
                range_mask = cv2.inRange(rgb_image, np.array(lower), np.array(upper))
                mask = cv2.bitwise_or(mask, range_mask)
        elif operation == "intersection":
            # 多个范围的交集
            mask = np.ones(rgb_image.shape[:2], dtype=np.uint8) * 255
            for lower, upper in ranges:
                range_mask = cv2.inRange(rgb_image, np.array(lower), np.array(upper))
                mask = cv2.bitwise_and(mask, range_mask)
        else:
            raise ValueError("operation 必须是 'union' 或 'intersection'")
        
        return mask
    
    def get_combined_mask(self, rgb_image: np.ndarray, 
                         color_names: List[str],
                         operation: str = "union") -> np.ndarray:
        """
        生成多个颜色组合的掩膜
        
        参数:
            rgb_image: RGB 格式的图像
            color_names: 颜色名称列表
            operation: 多个颜色之间的操作方式，"union"（并集）或 "intersection"（交集）
            
        返回:
            组合颜色掩膜（二值图像）
        """
        if not color_names:
            raise ValueError("颜色名称列表不能为空")
        
        masks = []
        for name in color_names:
            if name not in self.color_ranges:
                raise ValueError(f"未找到颜色范围: {name}")
            mask = self.get_color_mask(rgb_image, name, "union")
            masks.append(mask)
        
        if operation == "union":
            # 多个颜色的并集
            combined_mask = np.zeros(rgb_image.shape[:2], dtype=np.uint8)
            for mask in masks:
                combined_mask = cv2.bitwise_or(combined_mask, mask)
        elif operation == "intersection":
            # 多个颜色的交集
            combined_mask = np.ones(rgb_image.shape[:2], dtype=np.uint8) * 255
            for mask in masks:
                combined_mask = cv2.bitwise_and(combined_mask, mask)
        else:
            raise ValueError("operation 必须是 'union' 或 'intersection'")
        
        return combined_mask
    
    def get_color_distribution(self, rgb_image: np.ndarray) -> Tuple[np.ndarray, Dict]:
        """
        获取 RGB 图像的颜色分布
        
        参数:
            rgb_image: RGB 格式的图像
            
        返回:
            hist: 3D 直方图
            color_stats: 颜色分布统计信息
        """
        # 计算 3D 直方图
        hist = cv2.calcHist([rgb_image], [0, 1, 2], None, 
                           [self.r_bins, self.g_bins, self.b_bins], 
                           self.r_ranges + self.g_ranges + self.b_ranges)
        
        # 计算每个像素的量化颜色索引
        r_quantized = (rgb_image[:, :, 0] / (256 / self.r_bins)).astype(int)
        g_quantized = (rgb_image[:, :, 1] / (256 / self.g_bins)).astype(int)
        b_quantized = (rgb_image[:, :, 2] / (256 / self.b_bins)).astype(int)
        
        # 限制在有效范围内
        r_quantized = np.clip(r_quantized, 0, self.r_bins - 1)
        g_quantized = np.clip(g_quantized, 0, self.g_bins - 1)
        b_quantized = np.clip(b_quantized, 0, self.b_bins - 1)
        
        # 统计颜色分布
        color_indices = list(zip(r_quantized.flatten(), g_quantized.flatten(), b_quantized.flatten()))
        color_counter = Counter(color_indices)
        
        # 计算总像素数
        total_pixels = rgb_image.shape[0] * rgb_image.shape[1]
        
        # 转换为比例
        color_stats = {}
        for color, count in color_counter.items():
            color_stats[color] = count / total_pixels
        
        return hist, color_stats
    
    def get_dominant_colors(self, rgb_image: np.ndarray, top_k: int = 5) -> List[Tuple[Tuple[int, int, int], float, str]]:
        """
        获取图像中占比最多的几种颜色及其比例和颜色名称
        
        参数:
            rgb_image: RGB 格式的图像
            top_k: 返回前几种主要颜色
            
        返回:
            颜色列表，每个元素为 (RGB 值, 比例, 颜色名称)
        """
        _, color_stats = self.get_color_distribution(rgb_image)
        
        # 按比例排序
        sorted_colors = sorted(color_stats.items(), key=lambda x: x[1], reverse=True)
        
        # 转换为实际的 RGB 值和颜色名称
        dominant_colors = []
        for (r_idx, g_idx, b_idx), proportion in sorted_colors[:top_k]:
            r_val = int((r_idx + 0.5) * (256 / self.r_bins))
            g_val = int((g_idx + 0.5) * (256 / self.g_bins))
            b_val = int((b_idx + 0.5) * (256 / self.b_bins))
            color_name = self.rgb_to_color_name(r_val, g_val, b_val)
            dominant_colors.append(((r_val, g_val, b_val), proportion, color_name))
        
        return dominant_colors
    
    def get_all_color_proportions(self, rgb_image: np.ndarray, threshold: float = 0.01) -> List[Tuple[Tuple[int, int, int], float, str]]:
        """
        获取图像中所有颜色成分的比例（超过阈值）
        
        参数:
            rgb_image: RGB 格式的图像
            threshold: 显示比例阈值，低于此值的不显示
            
        返回:
            颜色列表，每个元素为 (RGB 值, 比例, 颜色名称)
        """
        _, color_stats = self.get_color_distribution(rgb_image)
        
        # 筛选超过阈值的颜色
        filtered_colors = [(color, prop) for color, prop in color_stats.items() if prop >= threshold]
        
        # 按比例排序
        sorted_colors = sorted(filtered_colors, key=lambda x: x[1], reverse=True)
        
        # 转换为实际的 RGB 值和颜色名称
        color_proportions = []
        for (r_idx, g_idx, b_idx), proportion in sorted_colors:
            r_val = int((r_idx + 0.5) * (256 / self.r_bins))
            g_val = int((g_idx + 0.5) * (256 / self.g_bins))
            b_val = int((b_idx + 0.5) * (256 / self.b_bins))
            color_name = self.rgb_to_color_name(r_val, g_val, b_val)
            color_proportions.append(((r_val, g_val, b_val), proportion, color_name))
        
        return color_proportions
    
    def remove_colors(self, rgb_image: np.ndarray, 
                     color_names: Union[str, List[str]]) -> np.ndarray:
        """
        删除指定颜色（将其变为黑色）
        
        参数:
            rgb_image: RGB 格式的图像
            color_names: 颜色名称或颜色名称列表
            
        返回:
            处理后的 RGB 图像
        """
        if isinstance(color_names, str):
            color_names = [color_names]
        
        mask = self.get_combined_mask(rgb_image, color_names, "union")
        
        # 将目标颜色区域变为黑色
        result = rgb_image.copy()
        result[mask > 0] = [0, 0, 0]  # RGB 中的黑色是 (0, 0, 0)
        
        return result
    
    def keep_colors(self, rgb_image: np.ndarray, 
                   color_names: Union[str, List[str]]) -> np.ndarray:
        """
        只保留指定颜色（其他变为黑色）
        
        参数:
            rgb_image: RGB 格式的图像
            color_names: 颜色名称或颜色名称列表
            
        返回:
            处理后的 RGB 图像
        """
        if isinstance(color_names, str):
            color_names = [color_names]
        
        mask = self.get_combined_mask(rgb_image, color_names, "union")
        
        # 将非目标颜色区域变为黑色
        result = rgb_image.copy()
        result[mask == 0] = [0, 0, 0]  # RGB 中的黑色是 (0, 0, 0)
        
        return result
    
    def remove_color_by_value(self, rgb_image: np.ndarray, target_rgb: Tuple[int, int, int], 
                             tolerance: Tuple[int, int, int] = (30, 30, 30)) -> np.ndarray:
        """
        删除指定 RGB 值范围的颜色（将其变为黑色）
        
        参数:
            rgb_image: RGB 格式的图像
            target_rgb: 目标 RGB 颜色 (R, G, B)
            tolerance: 容忍度 (R_tol, G_tol, B_tol)
            
        返回:
            处理后的 RGB 图像
        """
        r, g, b = target_rgb
        r_tol, g_tol, b_tol = tolerance
        
        # 创建掩码
        lower_bound = np.array([max(0, r - r_tol), max(0, g - g_tol), max(0, b - b_tol)])
        upper_bound = np.array([min(255, r + r_tol), min(255, g + g_tol), min(255, b + b_tol)])
        
        mask = cv2.inRange(rgb_image, lower_bound, upper_bound)
        
        # 将目标颜色区域变为黑色
        result = rgb_image.copy()
        result[mask > 0] = [0, 0, 0]  # RGB 中的黑色是 (0, 0, 0)
        
        return result
    
    def keep_color_by_value(self, rgb_image: np.ndarray, target_rgb: Tuple[int, int, int], 
                           tolerance: Tuple[int, int, int] = (30, 30, 30)) -> np.ndarray:
        """

    def apply_mask(self, rgb_image: np.ndarray, 
                  mask: np.ndarray, 
                  inverse: bool = False) -> np.ndarray:
        """
        应用掩膜到图像
        
        参数:
            rgb_image: RGB 格式的图像
            mask: 二值掩膜
            inverse: 是否反转掩膜
            
        返回:
            应用掩膜后的 RGB 图像
        """
        if inverse:
            mask = cv2.bitwise_not(mask)
        
        result = rgb_image.copy()
        result[mask == 0] = [0, 0, 0]

    def load_image(self, image_path: str, convert_to_rgb: bool = True) -> np.ndarray:

import cv2

class ColorRange:

        self.lower = np.array(lower, dtype=np.uint8)
        self.upper = np.array(upper, dtype=np.uint8)
    
    def get_lower(self) -> np.ndarray:
        """获取下界数组"""
        return self.lower
    
    def get_upper(self) -> np.ndarray:
        """获取上界数组"""
        return self.upper
    
    def to_dict(self) -> dict:
        """转换为字典格式"""
        return {
            "lower": self.lower.tolist(),
            "upper": self.upper.tolist()
        }
    
    @classmethod
    def from_dict(cls, data: dict) -> 'ColorRange':
        """从字典创建对象"""
        return cls(data["lower"], data["upper"])


class BallDetector:
    """球体检测器主类"""
    
    def __init__(self, config_path: Optional[str] = None):
        """
        构造函数
        
        Args:
            config_path: 配置文件路径，如果为None则使用默认值
        """
        # 默认配置
        self.default_config = {
            "color_range": {
                "lower": [20, 100, 100],
                "upper": [30, 255, 255]
            },
            "min_radius": 10,
            "max_radius": 100,
            "min_contour_area": 100
        }
        
        # 加载配置
        if config_path:
            self.config = self.load_config(config_path)
        else:
            self.config = self.default_config
        
        # 初始化颜色范围
        self.color_range = ColorRange.from_dict(self.config["color_range"])
        
        # ROI区域 (x, y, width, height)，None表示使用整个图像
        self.roi = None
        
        # 检测结果
        self.detection_results = []
    
    def load_config(self, config_path: str) -> dict:
        """
        加载配置文件
        
        Args:
            config_path: 配置文件路径
            
        Returns:
            配置字典
        """
        try:
            with open(config_path, 'r') as f:
                config = json.load(f)
            
            # 合并配置，确保所有必要字段都存在
            merged_config = self.default_config.copy()
            merged_config.update(config)
            
            return merged_config
        except FileNotFoundError:
            print(f"配置文件 {config_path} 未找到，使用默认配置")
            return self.default_config
        except json.JSONDecodeError:
            print(f"配置文件 {config_path} 格式错误，使用默认配置")
            return self.default_config
    
    def set_roi(self, x: int, y: int, width: int, height: int):
        """
        设置ROI区域
        
        Args:
            x: ROI左上角x坐标
            y: ROI左上角y坐标
            width: ROI宽度
            height: ROI高度
        """
        self.roi = (x, y, width, height)
        print(f"ROI设置为: ({x}, {y}, {width}, {height})")
    
    def clear_roi(self):
        """清除ROI设置，使用整个图像"""
        self.roi = None
        print("ROI已清除")
    
    def process_frame(self, frame: np.ndarray) -> Tuple[np.ndarray, Optional[Tuple]]:
        """
        对单帧图像进行处理和球体检测
        
        Args:
            frame: 输入图像帧
            
        Returns:
            processed_frame: 处理后的图像帧
            ball_info: 球体信息 (x, y, radius)，如果未检测到则为None
        """
        # 创建副本以免修改原图像
        processed_frame = frame.copy()
        
        # 应用ROI
        if self.roi:
            x, y, w, h = self.roi
            roi_frame = frame[y:y+h, x:x+w]
        else:
            roi_frame = frame
            x, y = 0, 0
        
        # 转换为HSV颜色空间
        hsv = cv2.cvtColor(roi_frame, cv2.COLOR_BGR2HSV)
        
        # 创建颜色掩膜
        mask = cv2.inRange(hsv, self.color_range.lower, self.color_range.upper)
        
        # 形态学操作去除噪声
        kernel = np.ones((5, 5), np.uint8)
        mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
        mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
        
        # 查找轮廓
        contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        
        ball_info = None
        
        if contours:
            # 找到最大轮廓
            largest_contour = max(contours, key=cv2.contourArea)
            
            # 计算轮廓面积
            area = cv2.contourArea(largest_contour)
            
            if area > self.config["min_contour_area"]:
                # 拟合最小外接圆
                ((ball_x, ball_y), radius) = cv2.minEnclosingCircle(largest_contour)
                
                # 转换为整数坐标（考虑ROI偏移）
                ball_x, ball_y = int(ball_x) + x, int(ball_y) + y
                radius = int(radius)
                
                # 检查半径是否在合理范围内
                if self.config["min_radius"] <= radius <= self.config["max_radius"]:
                    # 绘制检测结果
                    cv2.circle(processed_frame, (ball_x, ball_y), radius, (0, 255, 0), 2)
                    cv2.circle(processed_frame, (ball_x, ball_y), 2, (0, 0, 255), 3)
                    cv2.putText(processed_frame, f"Radius: {radius}", 
                               (ball_x - 50, ball_y - radius - 10), 
                               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)
                    
                    ball_info = (ball_x, ball_y, radius)
        
        # 绘制ROI区域
        if self.roi:
            cv2.rectangle(processed_frame, (x, y), (x + w, y + h), (255, 0, 0), 2)
            cv2.putText(processed_frame, "ROI", (x, y - 10), 
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 0, 0), 1)
        
        return processed_frame, ball_info
    
    def process_video(self, video_path: str, output_path: Optional[str] = None):
        """
        对整个视频进行处理
        
        Args:
            video_path: 输入视频路径
            output_path: 输出视频路径，如果为None则不保存
        """
        # 打开视频
        cap = cv2.VideoCapture("res/output.avi")
        
        if not cap.isOpened():
            print(f"无法打开视频文件: {video_path}")
            return
        
        # 获取视频属性
        fps = int(cap.get(cv2.CAP_PROP_FPS))
        width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
        height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
        
        # 设置视频写入器
        writer = None
        if output_path:
            fourcc = cv2.VideoWriter_fourcc(*'XVID')
            writer = cv2.VideoWriter(output_path, fourcc, fps, (width, height))
        
        frame_count = 0
        detection_count = 0
        
        print(f"开始处理视频: {video_path}")
        print(f"视频信息: {width}x{height}, {fps} FPS")
        
        while True:
            ret, frame = cap.read()
            
            if not ret:
                break
            
            # 处理当前帧
            processed_frame, ball_info = self.process_frame(frame)
            
            # 记录检测结果
            if ball_info:
                detection_count += 1
                self.detection_results.append({
                    "frame": frame_count,
                    "position": (ball_info[0], ball_info[1]),
                    "radius": ball_info[2]
                })
            
            # 显示处理结果
            cv2.imshow('Ball Detection', processed_frame)
            
            # 写入输出视频
            if writer:
                writer.write(processed_frame)
            
            frame_count += 1
            
            # 按'q'退出
            if cv2.waitKey(1) & 0xFF == ord('q'):
                break
        
        # 释放资源
        cap.release()
        if writer:
            writer.release()
        cv2.destroyAllWindows()
        
        # 输出统计信息
        detection_rate = (detection_count / frame_count) * 100 if frame_count > 0 else 0
        print(f"处理完成: 总帧数 {frame_count}, 检测到球体 {detection_count} 次")
        print(f"检测率: {detection_rate:.2f}%")
    
    def save_detection_results(self, output_path: str):
        """保存检测结果到JSON文件"""
        results = {
            "config": self.config,
            "detections": self.detection_results,
            "total_frames": len(self.detection_results)
        }
        
        with open(output_path, 'w') as f:
            json.dump(results, f, indent=2)
        
        print(f"检测结果已保存到: {output_path}")

def main():
    """主函数"""
    parser = argparse.ArgumentParser(description='球体检测器')
    parser.add_argument('--video', type=str, required=True, help='输入视频路径')
    parser.add_argument('--config', type=str, help='配置文件路径')
    parser.add_argument('--output', type=str, help='输出视频路径')
    parser.add_argument('--roi', type=str, help='ROI区域 (格式: x,y,width,height)')
    parser.add_argument('--create-config', action='store_true', help='创建示例配置文件')
    
    args = parser.parse_args()
    
    if args.create_config:
        create_sample_config()
        return
    
    # 创建检测器
    detector = BallDetector(args.config)
    
    # 设置ROI
    if args.roi:
        try:
            x, y, w, h = map(int, args.roi.split(','))
            detector.set_roi(x, y, w, h)
        except ValueError:
            print("ROI格式错误，请使用: x,y,width,height")
            return
    
    # 处理视频
    detector.process_video(args.video, args.output)
    
    # 保存检测结果
    if detector.detection_results:
        detector.save_detection_results("detection_results.json")


if __name__ == "__main__":
    main()
